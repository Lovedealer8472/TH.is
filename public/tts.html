Integrate in minutes. Async job model — fetch audio when it's ready.

import requests, time

API = "https://api.tolvuhvislarinn.is"
KEY = "tvh_your_key_here"
HEADERS = {"X-Api-Key": KEY}

# 1. Enqueue
r = requests.post(f"{API}/synthesize", headers=HEADERS, json={
    "text": "Góðan daginn! Hvernig líður þér?",
    "voice": "ugla",
})
job_id = r.json()["job_id"]

# 2. Poll
while True:
    status = requests.get(f"{API}/jobs/{job_id}", headers=HEADERS).json()
    if status["status"] == "done": break
    time.sleep(2)

# 3. Download
audio = requests.get(f"{API}/audio/{job_id}", headers=HEADERS)
open("output.wav", "wb").write(audio.content)

# 1. Enqueue synthesis
curl -X POST https://api.tolvuhvislarinn.is/synthesize \
  -H "X-Api-Key: tvh_your_key" \
  -H "Content-Type: application/json" \
  -d '{"text": "Halló heimur!", "voice": "ugla"}'

# 2. Check status
curl https://api.tolvuhvislarinn.is/jobs/{job_id} \
  -H "X-Api-Key: tvh_your_key"

# 3. Download audio
curl https://api.tolvuhvislarinn.is/audio/{job_id} \
  -H "X-Api-Key: tvh_your_key" \
  -o output.wav

async function synthesize(text, voice = "ugla") {
  const API = "https://api.tolvuhvislarinn.is";
  const headers = { "X-Api-Key": "tvh_your_key", "Content-Type": "application/json" };

  // Enqueue
  const { job_id } = await fetch(`${API}/synthesize`, {
    method: "POST", headers, body: JSON.stringify({ text, voice })
  }).then(r => r.json());

  // Poll
  while (true) {
    const s = await fetch(`${API}/jobs/${job_id}`, { headers }).then(r => r.json());
    if (s.status === "done") break;
    await new Promise(r => setTimeout(r, 2000));
  }

  // Get audio URL
  return `${API}/audio/${job_id}?key=tvh_your_key`;
}
